cmake_minimum_required(VERSION 3.30)
project(usd_mojo_bridge VERSION 0.2.0 LANGUAGES CXX)
find_package(pxr CONFIG REQUIRED)
set(USDM_SDK_RPATH)
foreach(SDK_PREFIX IN LISTS CMAKE_PREFIX_PATH)
  list(APPEND USDM_SDK_RPATH "${SDK_PREFIX}/lib")
endforeach()
add_library(usd_mojo_bridge SHARED bridge.cpp)
target_compile_features(usd_mojo_bridge PRIVATE cxx_std_17)
# Retain useful relative diagnostic paths without exposing local checkout/SDK prefixes.
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(usd_mojo_bridge PRIVATE
    "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=native")
  foreach(SDK_PREFIX IN LISTS CMAKE_PREFIX_PATH)
    target_compile_options(usd_mojo_bridge PRIVATE "-ffile-prefix-map=${SDK_PREFIX}=external-sdk")
  endforeach()
  # Imported SDK paths may retain a Pixi symlink instead of its resolved prefix.
  foreach(SDK_INCLUDE_DIR IN LISTS PXR_INCLUDE_DIRS)
    target_compile_options(usd_mojo_bridge PRIVATE
      "-ffile-prefix-map=${SDK_INCLUDE_DIR}=openusd-sdk/include")
  endforeach()
endif()
target_compile_options(usd_mojo_bridge PRIVATE -Wall -Wextra -Werror)
# GCC 12 ignores a scoped -Wcpp pragma for libstdc++'s legacy hash header.
# Keep that SDK preprocessor warning visible; ordinary C++ warnings stay fatal.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 13)
  target_compile_options(usd_mojo_bridge PRIVATE -Wno-error=cpp)
endif()
target_include_directories(usd_mojo_bridge PRIVATE ${PXR_INCLUDE_DIRS})
target_include_directories(usd_mojo_bridge SYSTEM PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/openusd)
target_link_libraries(usd_mojo_bridge PRIVATE usd usdGeom usdPhysics usdShade usdSkel usdUtils usdValidation sdf ar plug tf gf vt ts)
# The Conda SDK is built with Python support. Standalone C++/Mojo hosts must
# provide its CPython symbols even though this bridge never calls the Python API.
if(TARGET Python3::Python)
  target_link_libraries(usd_mojo_bridge PRIVATE Python3::Python)
endif()
set_target_properties(usd_mojo_bridge PROPERTIES
  CXX_VISIBILITY_PRESET default VISIBILITY_INLINES_HIDDEN NO
  BUILD_RPATH "${USDM_SDK_RPATH}")

# USD compares RTTI identity for asset/timecode arrays across shared libraries.
# Their template typeinfo must remain coalescible on macOS libc++.

if(APPLE)
  set_target_properties(usd_mojo_bridge PROPERTIES INSTALL_RPATH "@loader_path")
else()
  # Let the consumer's SDK RPATH reach dependencies of this installed bridge.
  # A bridge DT_RUNPATH would prevent that inherited search on glibc.
  target_link_options(usd_mojo_bridge PRIVATE -Wl,--disable-new-dtags)
  set_target_properties(usd_mojo_bridge PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS usd_mojo_bridge LIBRARY DESTINATION lib)

option(USD_MOJO_TESTS "Build native exception-boundary regression" OFF)
if(USD_MOJO_TESTS)
  add_executable(diagnostic_ownership ../tests/diagnostic_ownership.cpp)
  target_compile_features(diagnostic_ownership PRIVATE cxx_std_17)
  target_compile_options(diagnostic_ownership PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 13)
    target_compile_options(diagnostic_ownership PRIVATE -Wno-error=cpp)
  endif()
  target_include_directories(diagnostic_ownership PRIVATE ${PXR_INCLUDE_DIRS})
  target_include_directories(diagnostic_ownership SYSTEM PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/openusd)
  target_link_libraries(diagnostic_ownership PRIVATE usd usdGeom usdPhysics usdShade usdSkel usdUtils usdValidation sdf ar plug tf gf vt ts)
  if(TARGET Python3::Python)
    target_link_libraries(diagnostic_ownership PRIVATE Python3::Python)
  endif()
  if(NOT APPLE)
    target_link_directories(diagnostic_ownership PRIVATE "${CMAKE_PREFIX_PATH}/lib")
    target_link_options(diagnostic_ownership PRIVATE -Wl,--disable-new-dtags)
  endif()
  set_target_properties(diagnostic_ownership PROPERTIES BUILD_RPATH "${USDM_SDK_RPATH}")
endif()

option(USD_MOJO_IMAGING "Build the optional Hydra imaging library" OFF)
option(USD_MOJO_GL_TEST_CONTEXT "Build a process-scoped OpenGL context for imaging tests" OFF)
if(USD_MOJO_IMAGING)
  find_package(OpenGL REQUIRED)
  add_library(usd_mojo_imaging SHARED imaging.cpp)
  target_compile_features(usd_mojo_imaging PRIVATE cxx_std_17)
  target_compile_options(usd_mojo_imaging PRIVATE -Wall -Wextra -Werror
    "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=native")
  foreach(SDK_PREFIX IN LISTS CMAKE_PREFIX_PATH)
    target_compile_options(usd_mojo_imaging PRIVATE "-ffile-prefix-map=${SDK_PREFIX}=external-sdk")
  endforeach()
  foreach(SDK_INCLUDE_DIR IN LISTS PXR_INCLUDE_DIRS)
    target_compile_options(usd_mojo_imaging PRIVATE "-ffile-prefix-map=${SDK_INCLUDE_DIR}=openusd-sdk/include")
  endforeach()
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 13)
    target_compile_options(usd_mojo_imaging PRIVATE -Wno-error=cpp)
  endif()
  target_include_directories(usd_mojo_imaging PRIVATE ${PXR_INCLUDE_DIRS})
  target_link_libraries(usd_mojo_imaging PRIVATE usd_mojo_bridge usdImagingGL hgi hd usdGeom usd tf gf)
  if(TARGET Python3::Python)
    target_link_libraries(usd_mojo_imaging PRIVATE Python3::Python)
  endif()
  set_target_properties(usd_mojo_imaging PROPERTIES BUILD_RPATH "${USDM_SDK_RPATH}")
  if(APPLE)
    set_target_properties(usd_mojo_imaging PROPERTIES INSTALL_RPATH "@loader_path")
  else()
    target_link_options(usd_mojo_imaging PRIVATE -Wl,--disable-new-dtags)
    set_target_properties(usd_mojo_imaging PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
  install(TARGETS usd_mojo_imaging LIBRARY DESTINATION lib)
  if(USD_MOJO_GL_TEST_CONTEXT AND NOT APPLE)
    add_library(usd_mojo_test_context SHARED ../tests/imaging_context.cpp)
    target_compile_features(usd_mojo_test_context PRIVATE cxx_std_17)
    target_include_directories(usd_mojo_test_context SYSTEM PRIVATE ${PXR_INCLUDE_DIRS})
    target_link_libraries(usd_mojo_test_context PRIVATE glf garch)
    if(TARGET Python3::Python)
      target_link_libraries(usd_mojo_test_context PRIVATE Python3::Python)
    endif()
    set_target_properties(usd_mojo_test_context PROPERTIES BUILD_RPATH "${USDM_SDK_RPATH}")
  endif()
endif()

option(USD_MOJO_MATERIALX "Build optional native MaterialX document/shader support" OFF)
if(USD_MOJO_MATERIALX)
  find_package(MaterialX 1.39.5 EXACT CONFIG REQUIRED)
  if(APPLE)
    enable_language(OBJCXX)
    set_source_files_properties(materialx.cpp PROPERTIES LANGUAGE OBJCXX)
  endif()
  add_library(usd_mojo_materialx SHARED materialx.cpp)
  target_compile_features(usd_mojo_materialx PRIVATE cxx_std_17)
  target_compile_options(usd_mojo_materialx PRIVATE -Wall -Wextra -Werror
    "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=native")
  get_target_property(MX_INCLUDES MaterialXCore INTERFACE_INCLUDE_DIRECTORIES)
  foreach(MX_INCLUDE IN LISTS MX_INCLUDES)
    target_compile_options(usd_mojo_materialx PRIVATE "-ffile-prefix-map=${MX_INCLUDE}=materialx-sdk/include")
  endforeach()
  target_link_libraries(usd_mojo_materialx PRIVATE MaterialXCore MaterialXFormat MaterialXGenShader MaterialXGenGlsl MaterialXGenMsl)
  if(APPLE)
    target_link_libraries(usd_mojo_materialx PRIVATE MaterialXRenderMsl)
    set_target_properties(usd_mojo_materialx PROPERTIES INSTALL_RPATH "@loader_path")
  else()
    set_target_properties(usd_mojo_materialx PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
  install(TARGETS usd_mojo_materialx LIBRARY DESTINATION lib)
endif()
